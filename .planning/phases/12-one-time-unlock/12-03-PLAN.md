JS contract (must match frontend/lib/native/store/store-plugin.ts from plan 12-04 exactly):
  getProduct({ productId }) -> { productId, displayName, localizedPrice, price, currencyCode }   // localizedPrice = product.displayPrice; price = product.price as String (Decimal description), never a float
  purchase({ productId }) -> { status: 'success', jws, originalTransactionId, transactionId } | { status: 'cancelled' } | { status: 'pending' }
     .unverified -> reject code "UNVERIFIED"; product not found -> reject code "PRODUCT_NOT_FOUND"; StoreKit error -> reject code "PURCHASE_FAILED"
  currentEntitlements({ productId }) -> { transactions: [{ jws, originalTransactionId, transactionId }] }   // verified only, matching productId, no revoked (revocationDate == nil)
  sync() -> {}    // AppStore.sync(); reject "SYNC_FAILED" on error, "SYNC_CANCELLED" on userCancelled
  finish({ transactionId }) -> {}   // search Transaction.unfinished then Transaction.currentEntitlements for tx.id == transactionId, await tx.finish(); resolve even if not found
