frontend/lib/native/store/store-plugin.ts
  export const IAP_PRODUCT_ID = 'app.kitcha.mobile.full'
  export interface StoreProduct { productId: string; displayName: string; localizedPrice: string; price: string; currencyCode: string }
  export interface StoreTransaction { jws: string; originalTransactionId: string; transactionId: string }
  export type PurchaseResult = ({ status: 'success' } & StoreTransaction) | { status: 'cancelled' } | { status: 'pending' }
  export interface KumaunStorePlugin { getProduct(o: { productId: string }): Promise<StoreProduct>; purchase(o: { productId: string }): Promise<PurchaseResult>; currentEntitlements(o: { productId: string }): Promise<{ transactions: StoreTransaction[] }>; sync(): Promise<void>; finish(o: { transactionId: string }): Promise<void> }
  export async function getStorePlugin(): Promise<KumaunStorePlugin | null>   // null on web; memoized registerPlugin('KumaunStore') on native

frontend/lib/native/store/store-service.ts
  export interface StoreDeps { plugin: KumaunStorePlugin; verify(jws: string, idempotencyKey: string): Promise<VerifyResult>; newKey(): string; isOnline(): boolean }
  export type PurchaseOutcome = { kind: 'unlocked' } | { kind: 'revoked' } | { kind: 'cancelled' } | { kind: 'pending-approval' } | { kind: 'verify-pending' } | { kind: 'verify-failed' } | { kind: 'claimed-elsewhere' } | { kind: 'failed' }
  export type RestoreOutcome = { kind: 'restored' } | { kind: 'none' } | { kind: 'revoked' } | { kind: 'verify-pending' } | { kind: 'claimed-elsewhere' } | { kind: 'failed' }
  export async function loadProduct(deps): Promise<StoreProduct | null>
  export async function purchaseFull(deps): Promise<PurchaseOutcome>
  export async function restorePurchases(deps): Promise<RestoreOutcome>
  export async function recheckEntitlements(deps, serverStatus: EntitlementStatus): Promise<'verified' | 'skipped' | 'pending' | 'failed'>
