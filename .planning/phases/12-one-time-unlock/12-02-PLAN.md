Contract this plan creates (backend/src/modules/entitlements/apple-verifier.ts):
  export type AppleEnvironmentName = 'Sandbox' | 'Production' | 'Xcode'
  export interface AppleVerifierOptions { rootCertificates: readonly Buffer[]; bundleId: string; appAppleId?: number; onlineChecks: boolean; allowXcode: boolean }
  export interface VerifiedTransaction { transactionId: string; originalTransactionId: string; bundleId: string; productId: string; type: string; environment: AppleEnvironmentName; purchaseDate: Date; signedDate: Date; revocationDate: Date | null; revocationReason: number | null }
  export interface VerifiedNotification { notificationUUID: string; notificationType: string; subtype: string | null; signedDate: Date; environment: AppleEnvironmentName; transaction: VerifiedTransaction | null }
  export type AppleVerificationErrorKind = 'invalid' | 'retryable' | 'not_configured'
  export class AppleVerificationError extends Error { readonly kind: AppleVerificationErrorKind; readonly status?: number }
  export interface AppleVerifier { verifyTransaction(jws: string): Promise<VerifiedTransaction>; verifyNotification(signedPayload: string): Promise<VerifiedNotification> }
  export function createAppleVerifier(opts: AppleVerifierOptions): AppleVerifier
  export function appleRootCertificates(): Buffer[]   // decodes APPLE_ROOT_CA_G3_DER_BASE64

    - revocationDate present -> VerifiedTransaction.revocationDate is a Date and revocationReason a number
    - REFUND notification -> VerifiedNotification with nested transaction decoded; TEST notification without signedTransactionInfo -> transaction null
