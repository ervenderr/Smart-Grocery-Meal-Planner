  type DrainEvent =
    | { type: 'started' }
    | { type: 'applied'; op: OutboxOp; result: unknown }
    | { type: 'parked'; op: OutboxOp; error: LastError }
    | { type: 'retry-scheduled'; op: OutboxOp; at: number }
    | { type: 'offline' }
    | { type: 'auth-paused' }
    | { type: 'finished'; applied: number; parked: number }
  interface DrainWorker { kick(reason: string): void; whenIdle(): Promise<void>; isDraining(): boolean; dispose(): void }
  createDrainWorker({ store, transport, executors, connectivity, getUserId: () => string | null, now?: () => number, setTimer?: (fn, ms) => unknown, clearTimer?: (h) => void, onEvent?: (e: DrainEvent) => void }): DrainWorker
  Loop: single-flight (a kick during a run sets a rerun flag); stop when no user; check connectivity.isReachable() once per run start and after any network failure (emit offline and stop); pick nextReady; markInFlight; executor.run; on success complete + applied; on failure classify: ok -> complete + applied with result null; retry -> scheduleRetry(seq, { reason: 'transient', status, code }, now + nextDelayMs(attempts, { retryAfterMs }), countsAttempt), and on network failure call connectivity.reportNetworkFailure and stop; pause-auth -> store.pauseAuth(userId), emit auth-paused, stop; park -> store.park(seq, error), emit parked; missing executor -> park invalid. After the loop: arm one timer for the earliest next_attempt_at among queued ops (if any) that calls kick('backoff'); emit finished with counts for this run (11-13 uses it for one aggregated toast per drain).
- test-helpers/flaky-server.ts: createFlakyServer(): { transport: Transport; state: { shopping: Map; pantry: Map<id, { quantity: number }> }; script(step: FlakyStep): void; applyCount(key): number } where FlakyStep = 'drop-after-commit' | 'timeout' | { status: number; code?: string; retryAfterMs?: number } | 'network'. It mirrors the middleware: same key + same body after commit returns the stored response without re-applying; same key + different body -> 422 IDEMPOTENCY_KEY_REUSED; adjust clamps at 0; delete of missing -> 404.
